{{- /* github-stars.html Renders a star count chip from precomputed data (data/stars.json). Returns an empty string when no entry exists (graceful omission). Data is refreshed manually via scripts/update-stars.sh (requires GITHUB_TOKEN). Expects: a page context with params.github */ -}} {{- $out := "" -}} {{- with .Params.github -}} {{- $m := findRESubmatch `github\.com/([^/]+)/([^/?#]+)` . -}} {{- if $m -}} {{- $sub := index $m 0 -}} {{- $key := printf "%s/%s" (index $sub 1) (index $sub 2) -}} {{- with site.Data.stars -}} {{- with index . $key -}} {{- $n := int . -}} {{- if gt $n 0 -}} {{- if lt $n 1000 -}} {{- $out = printf "%d star" $n -}} {{- if gt $n 1 -}}{{- $out = printf "%d stars" $n -}}{{- end -}} {{- else if lt $n 1000000 -}} {{- $k := printf "%.1f" (div $n 1000.0) -}} {{- $k = strings.TrimSuffix ".0" $k -}} {{- $out = printf "%sK stars" $k -}} {{- else -}} {{- $m2 := printf "%.1f" (div $n 1000000.0) -}} {{- $m2 = strings.TrimSuffix ".0" $m2 -}} {{- $out = printf "%sM stars" $m2 -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $out -}}
